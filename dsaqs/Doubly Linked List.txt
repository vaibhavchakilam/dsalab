class DNode:
    def __init__(self,v): self.v,self.prev,self.next=v,None,None
class DLL:
    def __init__(self): self.head=None
    def insert_front(self,v):
        n=DNode(v); n.next=self.head
        if self.head: self.head.prev=n
        self.head=n
    def delete(self,v):
        cur=self.head
        while cur and cur.v!=v: cur=cur.next
        if not cur: return
        if cur.prev: cur.prev.next=cur.next
        else: self.head=cur.next
        if cur.next: cur.next.prev=cur.prev
    def traverse(self):
        cur=self.head
        while cur: print(cur.v); cur=cur.next
