class SegTree:
    def __init__(self,a):
        n=len(a); self.n=1
        while self.n<n: self.n*=2
        self.t=[0]*(2*self.n)
        for i,v in enumerate(a): self.t[self.n+i]=v
        for i in range(self.n-1,0,-1): self.t[i]=self.t[2*i]+self.t[2*i+1]
    def query(self,l,r): # inclusive l,r
        l+=self.n; r+=self.n; s=0
        while l<=r:
            if l%2==1: s+=self.t[l]; l+=1
            if r%2==0: s+=self.t[r]; r-=1
            l//=2; r//=2
        return s
# test
st=SegTree([1,2,3,4,5]); print(st.query(1,3))
