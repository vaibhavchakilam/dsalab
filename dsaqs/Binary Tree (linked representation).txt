# Binary Tree Node
class Node:
    def __init__(self, val):
        self.val = val
        self.left = None
        self.right = None

# Create nodes
root = Node(1)
root.left = Node(2)
root.right = Node(3)
root.left.left = Node(4)
root.left.right = Node(5)

# Display the tree using preorder (to check)
def preorder(root):
    if root:
        print(root.val, end=' ')
        preorder(root.left)
        preorder(root.right)

print("Preorder traversal of the binary tree:")
preorder(root)
